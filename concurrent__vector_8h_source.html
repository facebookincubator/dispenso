429 size_type capacity() const {
430 size_t cap = 2 * firstBucketLen_;
431 for (size_t b = 2; b < kMaxBuffers; ++b) {
432 if (!buffers_[b].load(std::memory_order_relaxed)) {
433 break;

444 void clear() {
445 auto binfo = bucketAndSubIndex(size_.load(std::memory_order_relaxed));
446 size_t len = binfo.bucketIndex;
447 size_t cap = binfo.bucketCapacity;
448 size_t b = binfo.bucket;
449 do {
450 T* buf = buffers_[b].load(std::memory_order_relaxed);
451 T* t = buf + len;
452
453 while (t != buf) {
454 --t;
455 t->~T();
456 }
457 cap >>= int{b > 1};
458 len = cap;
459 } while (b--);
460 size_.store(0, std::memory_order_release);
461 }

467 void shrink_to_fit() {
468 constexpr size_t kMaxExtra = 2;
469 auto binfo = bucketAndSubIndex(size_.load(std::memory_order_relaxed));
470
471 // We need to at least skip the first two buckets, since we have those as a single allocation.
472 size_t startBucket = std::max<size_t>(2, binfo.bucket + kMaxExtra);
473
474 for (size_t b = startBucket; b < kMaxBuffers; ++b) {
475 T* ptr = buffers_[b].load(std::memory_order_relaxed);
476 if (!ptr) {
477 break;
478 }
479 if (buffers_.shouldDealloc(b)) {
480 cv::dealloc<T>(ptr);
481 }
482 clearCachedPtr(b);
483 buffers_[b].store(nullptr, std::memory_order_release);
484 }
485 }

569 iterator erase(const_iterator pos) {
570 auto e = end();
571 if (e == pos) {
572 return e;
573 }
574 size_.fetch_sub(1, std::memory_order_relaxed);
575 --e;
576 if (e == pos) {
577 e->~T();
578 return e;
579 }
580 auto it = begin();
581 it += (pos - it);
582 it = std::move(pos + 1, const_iterator(e + 1), it);
583 // The last slot was vacated by the shift; destroy the moved-from object left there.
584 e->~T();
585 return it;
586 }

596 iterator erase(const_iterator first, const_iterator last) {
597 size_t len = std::distance(first, last);
598 auto it = begin();
599 size_t startIdx = first - it;
600 if (len == 0) {
601 return it + (startIdx + len);
602 }
603 it += startIdx;
604
605 auto e = end();
606 auto e_it = std::move(last, const_iterator(e), it);
607
608 // Destroy the vacated tail [e_it, end()): these are either moved-from or, when the tail was
609 // shorter than the erased range, erased elements that nothing was moved into.
610 while (e != e_it) {
611 --e;
612 e->~T();
613 }
614 size_.fetch_sub(len, std::memory_order_relaxed);
615 return e_it;
616 }

641 template <typename... Args>
642 iterator emplace_back(Args&&... args) {
643 auto index = size_.fetch_add(1, std::memory_order_relaxed);
644 auto binfo = bucketAndSubIndex(index);
645
646 allocateBuffer(binfo);
647
648 iterator ret{this, index, binfo};
649
650 if (Traits::kIteratorPreferSpeed) {
651 new (&*ret) T(std::forward<Args>(args)...);
652 } else {
653 new (buffers_[binfo.bucket] + binfo.bucketIndex) T(std::forward<Args>(args)...);
654 }
655
656 return ret;
657 }

1094 DISPENSO_INLINE cv::BucketInfo bucketAndSubIndexForIndex(size_t index) const {
1095#if defined(_MSC_VER) || defined(__aarch64__) || defined(_M_ARM64)
1096 // Branching fast path — benefits MSVC and ARM where branch prediction handles
1097 // sequential access patterns well, avoiding log2/division for the common case.
1098 if (index < firstBucketLen_) {
1099 return {0, index, firstBucketLen_};
1100 }
1101
1102 size_t l2idx = detail::log2(index);
1103 size_t bucket = (l2idx + 1) - firstBucketShift_;
1104 size_t bucketCapacity = size_t{1} << l2idx;
1105 size_t bucketIndex = index - bucketCapacity;
1106
1107 return {bucket, bucketIndex, bucketCapacity};
1108#else
1109 return bucketAndSubIndex(index);
1110#endif // _MSC_VER || __aarch64__ || _M_ARM64
1111 }
1112
1113 DISPENSO_INLINE cv::BucketInfo bucketAndSubIndex(size_t index) const {
1114 size_t l2idx = detail::log2(index | 1);
1115 size_t bucket = (l2idx + 1) - firstBucketShift_;
1116 size_t bucketCapacity = size_t{1} << l2idx;
1117 size_t bucketIndex = index - bucketCapacity;
1118
1119 bucket = index < firstBucketLen_ ? 0 : bucket;
1120 bucketIndex = index < firstBucketLen_ ? index : bucketIndex;
1121 bucketCapacity = index < firstBucketLen_ ? firstBucketLen_ : bucketCapacity;
1122
1123 return {bucket, bucketIndex, bucketCapacity};
1124 }

1221 DISPENSO_INLINE T* cachedBuffer(size_t bucket) const {
1222#if DISPENSO_HAS_CACHED_PTRS
1223 return cachedPtrs_[bucket];
1224#else

1227 // transitively carries the buffer pointer visibility (which was release-stored during
1228 // allocation). This matches the original pre-cache memory ordering and avoids the cost of
1229 // ldar (load-acquire) on AArch64.
