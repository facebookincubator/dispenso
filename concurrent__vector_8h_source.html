441 size_t cap = 2 * firstBucketLen_;
442 for (size_t b = 2; b < kMaxBuffers; ++b) {

455 void clear() {
456#if defined(__cpp_exceptions)
457 if (DISPENSO_EXPECT(gaps_.load(std::memory_order_relaxed) != nullptr, false)) {
458 clearWithGaps();
459 return;
460 }
461#endif // __cpp_exceptions
462 auto binfo = bucketAndSubIndex(size_.load(std::memory_order_relaxed));
463 size_t len = binfo.bucketIndex;
464 size_t cap = binfo.bucketCapacity;
465 size_t b = binfo.bucket;
466 do {
467 T* buf = buffers_[b].load(std::memory_order_relaxed);
468 T* t = buf + len;
469
470 while (t != buf) {
471 --t;
472 t->~T();
473 }
474 cap >>= int{b > 1};
475 len = cap;
476 } while (b--);
477 size_.store(0, std::memory_order_release);
478 }

484 void shrink_to_fit() {
485 constexpr size_t kMaxExtra = 2;
486 auto binfo = bucketAndSubIndex(size_.load(std::memory_order_relaxed));
487
488 // We need to at least skip the first two buckets, since we have those as a single allocation.
489 size_t startBucket = std::max<size_t>(2, binfo.bucket + kMaxExtra);
490
491 for (size_t b = startBucket; b < kMaxBuffers; ++b) {
492 T* ptr = buffers_[b].load(std::memory_order_relaxed);
493 if (!ptr) {
494 break;
495 }
496 if (buffers_.shouldDealloc(b)) {
497 cv::dealloc<T>(ptr);
498 }
499 clearCachedPtr(b);
500 buffers_[b].store(nullptr, std::memory_order_release);
501 }
502 }

586 iterator erase(const_iterator pos) {
587 auto e = end();
588 if (e == pos) {
589 return e;
590 }
591 size_.fetch_sub(1, std::memory_order_relaxed);
592 --e;
593 if (e == pos) {
594 e->~T();
595 return e;
596 }
597 auto it = begin();
598 it += (pos - it);
599 it = std::move(pos + 1, const_iterator(e + 1), it);
600 // The last slot was vacated by the shift; destroy the moved-from object left there.
601 e->~T();
602 return it;
603 }
604
613 iterator erase(const_iterator first, const_iterator last) {
614 size_t len = std::distance(first, last);
615 auto it = begin();
616 size_t startIdx = first - it;
617 if (len == 0) {
618 return it + (startIdx + len);
619 }
620 it += startIdx;
621
622 auto e = end();
623 auto e_it = std::move(last, const_iterator(e), it);
624
625 // Destroy the vacated tail [e_it, end()): these are either moved-from or, when the tail was
626 // shorter than the erased range, erased elements that nothing was moved into.
627 while (e != e_it) {
628 --e;
629 e->~T();
630 }
631 size_.fetch_sub(len, std::memory_order_relaxed);
632 return e_it;
633 }

658 template <typename... Args>
659 iterator emplace_back(Args&&... args) {
660 auto index = size_.fetch_add(1, std::memory_order_relaxed);
661 auto binfo = bucketAndSubIndex(index);
662
663 allocateBuffer(binfo);
664
665 iterator ret{this, index, binfo};
666
667 T* slot = Traits::kIteratorPreferSpeed ? &*ret : buffers_[binfo.bucket] + binfo.bucketIndex;
668#if defined(__cpp_exceptions)
669 try {
670 new (slot) T(std::forward<Args>(args)...);
671 } catch (...) {
672 finishFailedGrowth(ret, 1);
673 throw;
674 }
675#else
676 new (slot) T(std::forward<Args>(args)...);
677#endif // __cpp_exceptions
678
679 return ret;
680 }

1117 DISPENSO_INLINE cv::BucketInfo bucketAndSubIndexForIndex(size_t index) const {
1118#if defined(_MSC_VER) || defined(__aarch64__) || defined(_M_ARM64)
1119 // Branching fast path — benefits MSVC and ARM where branch prediction handles
1120 // sequential access patterns well, avoiding log2/division for the common case.
1121 if (index < firstBucketLen_) {
1122 return {0, index, firstBucketLen_};
1123 }
1124
1125 size_t l2idx = detail::log2(index);
1126 size_t bucket = (l2idx + 1) - firstBucketShift_;
1127 size_t bucketCapacity = size_t{1} << l2idx;
1128 size_t bucketIndex = index - bucketCapacity;
1129
1130 return {bucket, bucketIndex, bucketCapacity};
1131#else
1132 return bucketAndSubIndex(index);
1133#endif // _MSC_VER || __aarch64__ || _M_ARM64
1134 }
1135
1136 DISPENSO_INLINE cv::BucketInfo bucketAndSubIndex(size_t index) const {
1137 size_t l2idx = detail::log2(index | 1);
1138 size_t bucket = (l2idx + 1) - firstBucketShift_;
1139 size_t bucketCapacity = size_t{1} << l2idx;
1140 size_t bucketIndex = index - bucketCapacity;
1141
1142 bucket = index < firstBucketLen_ ? 0 : bucket;
1143 bucketIndex = index < firstBucketLen_ ? index : bucketIndex;
1144 bucketCapacity = index < firstBucketLen_ ? firstBucketLen_ : bucketCapacity;
1145
1146 return {bucket, bucketIndex, bucketCapacity};
1147 }

1149 // Sets size_ to the number of elements constructed so far when it goes out of scope, including
1150 // when a T constructor throws. Used where the vector is filled from empty, not concurrently, so
1151 // that the destructor (or a later clear()) destroys exactly the elements that were constructed.

1190 // frees the gap records. Allocates nothing.
