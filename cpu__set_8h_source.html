235 static constexpr uint32_t kRefreshPeriod = 32;
236 static DISPENSO_THREAD_LOCAL int32_t cachedCpu = -1;
237 static DISPENSO_THREAD_LOCAL uint32_t counter = 0;
238 if ((counter++ & (kRefreshPeriod - 1)) == 0) {
