54class OnceFunction {
55 public:
59 OnceFunction() : ops_(&detail::EmptyOnceCallableOps<>::kOps) {}
60
68 template <typename F>
69 DISPENSO_REQUIRES(OnceCallableFunc<F>)
70 OnceFunction(F&& f) : ops_(detail::createOnceCallable(std::forward<F>(f), buf_)) {}
71
72 OnceFunction(const OnceFunction& other) = delete;
73
78 OnceFunction(OnceFunction&& other) noexcept {
79 moveFrom(other);
80 }
81
82 OnceFunction& operator=(OnceFunction&& other) noexcept {
83 if (this != &other) {
84#if defined DISPENSO_DEBUG
85 assert(
86 ops_->invoke == nullptr &&
87 "OnceFunction must be invoked or cleanupNotRun() before reassignment");
88#endif // DISPENSO_DEBUG
89 moveFrom(other);
90 }
91 return *this;
92 }
93
99 void cleanupNotRun() {
100#if defined DISPENSO_DEBUG
101 assert(ops_->invoke != nullptr && "Must not cleanup an invalid OnceFunction!");
102#endif
103 consumeOps()->invoke(buf_, false);
104 }
105
110 void operator()() const {
111#if defined DISPENSO_DEBUG
112 assert(ops_->invoke != nullptr && "Must not use OnceFunction more than once!");
113#endif // DISPENSO_DEBUG
114 consumeOps()->invoke(buf_, true);
115 }
116
117 private:
118 // 64 bytes total (one cache line): 56 bytes inline storage + 8 bytes operations ptr.
119 // Inline: buf_ holds the functor directly.
120 // Spill: buf_[0..7] holds a void* to pool-allocated storage.
121 // Moves are a 64-byte memcpy, except for inline functors that are not trivially copyable, which
122 // the operations' relocate entry move-constructs into place.
123 void moveFrom(OnceFunction& other) noexcept {
124 if (DISPENSO_EXPECT(other.ops_->relocate == nullptr, true)) {
125 std::memcpy(static_cast<void*>(this), &other, sizeof(OnceFunction));
126 } else {
127 ops_ = other.ops_;
128 ops_->relocate(other.buf_, buf_);
129 other.ops_ = &detail::EmptyOnceCallableOps<>::kOps;
130 }
131#if defined DISPENSO_DEBUG
132 other.ops_ = &detail::EmptyOnceCallableOps<>::kOps;
133#endif // DISPENSO_DEBUG
134 }
135
136 // Marks this object consumed before the call, because the invoked functor may free the memory
137 // holding this OnceFunction. Leaves no stale relocate entry for a later move to act on.
138 const detail::OnceCallableOps* consumeOps() const {
139 const detail::OnceCallableOps* ops = ops_;
140 ops_ = &detail::EmptyOnceCallableOps<>::kOps;
141 return ops;
142 }
143
144 alignas(64) mutable char buf_[detail::kOnceFunctionInlineSize];
145 mutable const detail::OnceCallableOps* ops_;
146
147 template <typename Result>
148 friend class detail::FutureBase;
149 template <typename Result>
150 friend class detail::FutureImplBase;
151};
