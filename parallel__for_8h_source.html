126template <typename IntegerT = ssize_t>
127struct ChunkedRange {
128 // We need to utilize 64-bit integers to avoid overflow, e.g. passing -2**30, 2**30 as int32 will
129 // result in overflow unless we cast to 64-bit. Note that if we have a range of e.g. -2**63+1 to
130 // 2**63-1, we cannot hold the result in an int64_t. We could in a uint64_t, but it is quite
131 // tricky to make this work. However, I do not expect ranges larger than can be held in int64_t
132 // since people want their computations to finish before the heat death of the sun (slight
133 // exaggeration).
134 using size_type = std::conditional_t<std::is_signed<IntegerT>::value, int64_t, uint64_t>;
135
136 struct Static {};
137 struct Auto {};
138 static constexpr IntegerT kStatic = std::numeric_limits<IntegerT>::max();
139
147 ChunkedRange(IntegerT s, IntegerT e, IntegerT c) : start(s), end(e), chunk(c) {}
154 ChunkedRange(IntegerT s, IntegerT e, Static) : ChunkedRange(s, e, kStatic) {}
162 ChunkedRange(IntegerT s, IntegerT e, Auto) : ChunkedRange(s, e, 0) {}
163
164 bool isStatic() const {
165 return chunk == kStatic;
166 }
167
168 bool isAuto() const {
169 return chunk == 0;
170 }
171
172 bool empty() const {
173 return end <= start;
174 }
175
176 size_type size() const {
177 return static_cast<size_type>(end) - start;
178 }

180 template <typename OtherInt>
181 std::tuple<size_type, size_type>
182 calcChunkSize(OtherInt numLaunched, bool oneOnCaller, size_type minChunkSize) const {
183 size_type workingThreads = static_cast<size_type>(numLaunched) + size_type{oneOnCaller};
184 assert(workingThreads > 0);
185
186 if (!chunk) {
187 size_type dynFactor = std::min<size_type>(16, size() / workingThreads);
188 size_type chunkSize;
189 do {
190 size_type roughChunks = dynFactor * workingThreads;
191 chunkSize = (size() + roughChunks - 1) / roughChunks;
192 --dynFactor;
193 } while (chunkSize < minChunkSize);
194 return {chunkSize, (size() + chunkSize - 1) / chunkSize};
195 } else if (chunk == kStatic) {
196 // This should never be called. The static distribution versions of the parallel_for
197 // functions should be invoked instead.
198 std::abort();
199 }
200 return {chunk, (size() + chunk - 1) / chunk};
201 }
202
203 IntegerT start;
204 IntegerT end;
205 IntegerT chunk;
206};

340template <
341 typename TaskSetT,
342 typename IntegerT,
343 typename F,
344 typename StateContainer,
345 typename StateGen>
346void parallel_for_staticImpl(
347 TaskSetT& taskSet,
348 StateContainer& states,
349 const StateGen& defaultState,
350 const ChunkedRange<IntegerT>& range,
351 F&& f,
352 ssize_t maxThreads,
353 bool wait,
354 bool reuseExistingState) {
355 using size_type = typename ChunkedRange<IntegerT>::size_type;
356
357 size_type numThreads = std::min<size_type>(taskSet.numPoolThreads() + 1, maxThreads);
358 // Reduce threads used if they exceed work to be done.
359 numThreads = std::min(numThreads, range.size());
360
361 detail::initStates(states, defaultState, static_cast<size_t>(numThreads), reuseExistingState);
362
363 auto chunking =
364 detail::staticChunkSize(static_cast<ssize_t>(range.size()), static_cast<ssize_t>(numThreads));
365 IntegerT chunkSize = static_cast<IntegerT>(chunking.ceilChunkSize);
366
367 bool perfectlyChunked = static_cast<size_type>(chunking.transitionTaskIndex) == numThreads;
368
369 // Helper: compute chunk [start, end) for a given task index.
370 auto chunkRange = [&](size_type idx) -> std::pair<IntegerT, IntegerT> {
371 size_type transIdx = perfectlyChunked ? numThreads : chunking.transitionTaskIndex;
372 IntegerT smallChunk = static_cast<IntegerT>(chunkSize - !perfectlyChunked);
373 IntegerT start;
374 if (idx < transIdx) {
375 IntegerT i = static_cast<IntegerT>(idx);
376 start = static_cast<IntegerT>(range.start + static_cast<IntegerT>(i * chunkSize));
377 } else {
378 IntegerT ti = static_cast<IntegerT>(transIdx);
379 IntegerT ri = static_cast<IntegerT>(idx - transIdx);
380 start = static_cast<IntegerT>(
381 range.start + static_cast<IntegerT>(ti * chunkSize) +
382 static_cast<IntegerT>(ri * smallChunk));
383 }
384 IntegerT end;
385 if (idx + 1 == numThreads) {
386 end = range.end;
387 } else if (idx < transIdx) {
388 end = static_cast<IntegerT>(start + chunkSize);
389 } else {
390 end = static_cast<IntegerT>(start + smallChunk);
391 }
392 return {start, end};
393 };
394
395 // Determine which chunk the calling thread should take for L2 locality.
396 // If the caller is a pool thread with a ring, it takes the chunk matching
397 // its ring index so that repeated parallel_for calls keep the same data
398 // in the caller's L2 cache. Otherwise takes the last chunk.
399 int32_t callerRing = wait ? detail::PerPoolPerThreadInfo::ringIndex(&taskSet.pool()) : -1;
400 size_type callerChunk = numThreads - 1;
401 if (callerRing >= 0 && static_cast<size_type>(callerRing) < numThreads) {
402 callerChunk = static_cast<size_type>(callerRing);
403 }
404
405 // Schedule N-1 chunks to workers (skipping callerChunk when wait=true).
406 // The generator remaps scheduler index to chunk index: indices below
407 // callerChunk map 1:1, indices >= callerChunk shift up by 1.
408 // This ensures ring i gets the data for chunk i (or chunk i+1 if i >= callerChunk),
409 // maintaining L2 locality for each worker thread.
410 size_type numToSchedule = wait ? numThreads - 1 : numThreads;
411
412 if (numToSchedule > 0) {
413 taskSet.scheduleBulk(static_cast<size_t>(numToSchedule), [&, callerChunk](size_t idx) {
414 size_type chunkIdx = static_cast<size_type>(idx);
415 if (wait && chunkIdx >= callerChunk) {
416 ++chunkIdx;
417 }
418
419 auto chunkBounds = chunkRange(chunkIdx);
420 IntegerT start = chunkBounds.first;
421 IntegerT end = chunkBounds.second;
422
423 auto stateIt = states.begin();
424 std::advance(stateIt, static_cast<ptrdiff_t>(chunkIdx));
425
426 return [it = stateIt, start, end, f]() {
427 auto recurseInfo = detail::PerPoolPerThreadInfo::parForRecurse();
428 f(*it, start, end);
429 };
430 });
431 }
432
433 if (wait) {
434 auto stateIt = states.begin();
435 std::advance(stateIt, static_cast<ptrdiff_t>(callerChunk));
436 auto callerBounds = chunkRange(callerChunk);
437 {
438 // Mark caller as inside parallel_for so nested parallel_for calls
439 // skip the ring fast path (same as worker tasks which have this guard
440 // in their lambda). Without this, the caller's nested parallel_for
441 // would push to rings that workers can't drain while they're busy
442 // with their own outer chunks.
443 auto recurseInfo = detail::PerPoolPerThreadInfo::parForRecurse();
444 f(*stateIt, callerBounds.first, callerBounds.second);
445 }
446 taskSet.wait();
447 }
448}

468template <typename IntegerT>
469ChunkSizingResult<IntegerT> adjustChunkSizing(
470 const ChunkedRange<IntegerT>& range,
471 typename ChunkedRange<IntegerT>::size_type maxThreads,
472 bool isStatic,
473 uint32_t minItemsPerChunk,
474 typename ChunkedRange<IntegerT>::size_type poolThreads,
475 bool wait) {
476 using size_type = typename ChunkedRange<IntegerT>::size_type;
477
478 // Step 1: never use more threads than could participate.
479 // Always count the caller (+1) even when wait=false, because the caller
480 // will eventually participate via TaskSet::wait()/destructor and can steal
481 // work items. This prevents noWait parallel_for on small pools from
482 // degenerating to serial inline execution.
483 maxThreads = std::min<size_type>(maxThreads, poolThreads + 1);
484
485 if (minItemsPerChunk > 1) {
486 // Step 2a: reduce threads so each gets at least minItemsPerChunk items
487 size_type maxWorkers = range.size() / minItemsPerChunk;
488 if (maxWorkers < maxThreads) {
489 maxThreads = maxWorkers;
490 }
491 // Step 2b: if dynamic chunks would still be too small, use static scheduling
492 if (maxThreads > 0 && range.size() / (maxThreads + wait) < minItemsPerChunk && range.isAuto()) {
493 isStatic = true;
494 }
495 } else if (range.size() <= poolThreads + wait) {
496 // Step 3: fewer items than threads — static is better (no atomic overhead)
497 if (range.isAuto()) {
498 isStatic = true;
499 } else if (!range.isStatic()) {
500 maxThreads = range.size() - wait;
501 }
502 }
503
504 return {maxThreads, isStatic};
505}

507// Per-group range for decoupled-atomic dynamic parallel_for. Each group of
508// threads shares an L3-local atomic index over a contiguous sub-range of
509// chunks, eliminating cross-CCD cache-line bouncing on many-core machines.
510struct alignas(kCacheLineSize) GroupRange {
511 std::atomic<size_t> index{0};
512 size_t startChunk{0};
513 size_t numGroupChunks{0};
514};

536template <
537 typename TaskSetT,
538 typename IntegerT,
539 typename F,
540 typename StateContainer,
541 typename IndexRef,
542 typename ExitAction>
543void parallel_for_dynamicImpl(
544 TaskSetT& taskSet,
545 StateContainer& states,
546 IntegerT start,
547 IntegerT end,
548 F&& f,
549 size_t numToLaunch,
550 typename ChunkedRange<IntegerT>::size_type chunkSize,
551 typename ChunkedRange<IntegerT>::size_type numChunks,
552 IndexRef& index,
553 ExitAction exitAction,
554 bool wait) {
555 size_t totalWorkers = numToLaunch + (wait ? 1 : 0);
556
557 // Use the L3 cache group count (one per CCD on AMD, one per tile/SNC
558 // cluster on Intel) when available, so the per-group atomic lands in each
559 // CCD's L3. Falls back to a heuristic (~16 threads per group) on platforms
560 // where sysfs/topology detection returns nothing (e.g. macOS).
561 size_t l3Groups = CpuSet::l3CacheGroups().size();
562 size_t effectiveGroups;
563 if (l3Groups > 1 && totalWorkers > 16) {
564 effectiveGroups = std::min(l3Groups, totalWorkers);
565 } else {
566 effectiveGroups = std::max<size_t>(1, (totalWorkers + 15) / 16);
567 }
568
569 // Single group: use the original shared-index path with no extra overhead.
570 if (effectiveGroups <= 1) {
571 auto worker = [start, end, &index, f, chunkSize, numChunks, exitAction](auto& s) {
572 auto recurseInfo = detail::PerPoolPerThreadInfo::parForRecurse();
573 while (true) {
574 auto cur = index.fetch_add(1, std::memory_order_relaxed);
575 if (cur >= numChunks) {
576 exitAction(cur);
577 break;
578 }
579 auto sidx = static_cast<IntegerT>(start + cur * chunkSize);
580 if (cur + 1 == numChunks) {
581 f(s, sidx, end);
582 } else {
583 f(s, sidx, static_cast<IntegerT>(sidx + chunkSize));
584 }
585 }
586 };
587
588 {
589 auto it = states.begin();
590 for (size_t i = 0; i < static_cast<size_t>(numToLaunch); ++i) {
591 taskSet.schedule([&s = *it++, worker]() { worker(s); });
592 }
593 }
594
595 if (wait) {
596 auto it = states.begin();
597 std::advance(it, static_cast<ptrdiff_t>(numToLaunch));
598 worker(*it);
599 taskSet.wait();
600 }
601 return;
602 }
603
604 // Multi-group path: partition chunks among effectiveGroups groups, each with
605 // its own cache-line-aligned atomic index.
606 //
607 // The GroupRange array and exit counter must outlive all workers. In the
608 // wait path the caller blocks until completion, so stack ownership suffices.
609 // In the no-wait path the function returns immediately, so we heap-allocate
610 // and have the last worker free the memory alongside the caller's exitAction.
611
612 // Header stored at the front of a single aligned allocation that holds the
613 // exit counter followed by the GroupRange array.
614 struct alignas(kCacheLineSize) GroupBlock {
615 std::atomic<size_t> exitCounter{0};
616 size_t numGroups;
617 size_t totalWorkers;
618 bool heapOwned;
619
620 GroupRange* ranges() {
621 auto* base = reinterpret_cast<char*>(this);
622 uintptr_t off = detail::alignToCacheLine(sizeof(GroupBlock));
623 return reinterpret_cast<GroupRange*>(base + off);
624 }
625 };
626
627 size_t blockBytes =
628 detail::alignToCacheLine(sizeof(GroupBlock)) + sizeof(GroupRange) * effectiveGroups;
629
630 // Both the wait and no-wait paths heap-own the block; the last worker to
631 // finish frees it. A thread-local buffer-reuse optimization for the wait
632 // path was intentionally not used here: it is unsafe under nested multi-group
633 // parallel_for on the same thread, because the caller runs its own chunk
634 // (worker(*it) below) -- which may re-enter here -- BEFORE taskSet.wait()
635 // returns, while this call's other-thread workers are still reading the
636 // block. The per-call allocation is negligible relative to the >16-thread
637 // work on this path; revisit only if it shows up in a profile.
638 void* blockMem = detail::alignedMalloc(blockBytes, kCacheLineSize);
639 bool heapOwned = true;
640
641 auto* block = new (blockMem) GroupBlock{};
642 block->numGroups = effectiveGroups;
643 block->totalWorkers = totalWorkers;
644 block->heapOwned = heapOwned;
645
646 GroupRange* groupRanges = block->ranges();
647 for (size_t g = 0; g < effectiveGroups; ++g) {
648 new (&groupRanges[g]) GroupRange{};
649 }
650
651 size_t baseChunks = static_cast<size_t>(numChunks) / effectiveGroups;
652 size_t extraChunks = static_cast<size_t>(numChunks) % effectiveGroups;
653 size_t chunkOffset = 0;
654 for (size_t g = 0; g < effectiveGroups; ++g) {
655 size_t gc = baseChunks + (g < extraChunks ? 1 : 0);
656 groupRanges[g].startChunk = chunkOffset;
657 groupRanges[g].numGroupChunks = gc;
658 chunkOffset += gc;
659 }
660
661 auto worker = [start, end, block, f, chunkSize, numChunks, exitAction](auto& s, size_t groupIdx) {
662 auto recurseInfo = detail::PerPoolPerThreadInfo::parForRecurse();
663 auto& gr = block->ranges()[groupIdx];
664 // Snapshot block-owned metadata BEFORE the release increment below. Once a
665 // worker performs exitCounter.fetch_add, the worker that observes the final
666 // count frees `block`, so `block` must not be dereferenced afterwards.
667 // Reading block->totalWorkers in the `prev + 1 == ...` check after the
668 // increment is a use-after-free race against that free (caught by TSAN).
669 const size_t totalWorkersLocal = block->totalWorkers;
670 const bool owned = block->heapOwned;
671 while (true) {
672 auto cur = gr.index.fetch_add(1, std::memory_order_relaxed);
673 if (cur >= gr.numGroupChunks) {
674 break;
675 }
676 auto globalChunk =
677 static_cast<typename ChunkedRange<IntegerT>::size_type>(gr.startChunk + cur);
678 auto sidx = static_cast<IntegerT>(start + globalChunk * chunkSize);
679 if (globalChunk + 1 == numChunks) {
680 f(s, sidx, end);
681 } else {
682 f(s, sidx, static_cast<IntegerT>(sidx + chunkSize));
683 }
684 }
685 // The acq_rel increment orders every worker's prior block accesses (the gr
686 // loop above) before the last worker's free. After it, only the last worker
687 // touches block, and only to free it -- no peer reads block past this point.
688 auto prev = block->exitCounter.fetch_add(1, std::memory_order_acq_rel);
689 if (prev + 1 == totalWorkersLocal) {
690 exitAction(numChunks + static_cast<decltype(numChunks)>(totalWorkersLocal) - 1);
691 if (owned) {
692 detail::alignedFree(block);
693 }
694 }
695 };
696
697 {
698 auto it = states.begin();
699 for (size_t i = 0; i < numToLaunch; ++i) {
700 size_t gIdx = (i * effectiveGroups) / totalWorkers;
701 taskSet.schedule([&s = *it++, worker, gIdx]() { worker(s, gIdx); });
702 }
703 }
704
705 if (wait) {
706 auto it = states.begin();
707 std::advance(it, static_cast<ptrdiff_t>(numToLaunch));
708 size_t gIdx = (numToLaunch * effectiveGroups) / totalWorkers;
709 if (gIdx >= effectiveGroups) {
710 gIdx = effectiveGroups - 1;
711 }
712 worker(*it, gIdx);
713 taskSet.wait();
714 }
715}

735template <
736 typename TaskSetT,
737 typename IntegerT,
738 typename F,
739 typename StateContainer,
740 typename StateGen>
741void parallel_for(
742 TaskSetT& taskSet,
743 StateContainer& states,
744 const StateGen& defaultState,
745 const ChunkedRange<IntegerT>& range,
746 F&& f,
747 ParForOptions options = {}) {
748 if (range.empty()) {
749 if (options.wait) {
750 taskSet.wait();
751 }
752 return;
753 }
754
755 using size_type = typename ChunkedRange<IntegerT>::size_type;
756
757 uint32_t minItemsPerChunk = std::max<uint32_t>(1, options.minItemsPerChunk);
758 size_type maxThreads = std::max<int32_t>(options.maxThreads, 1);
759 bool isStatic = range.isStatic();
760
761 const size_type N = taskSet.numPoolThreads();
762 if (N == 0 || !options.maxThreads || range.size() <= minItemsPerChunk ||
763 detail::PerPoolPerThreadInfo::isParForRecursive(&taskSet.pool())) {
764 detail::initStates(states, defaultState, 1, options.reuseExistingState);
765 f(*states.begin(), range.start, range.end);
766 if (options.wait) {
767 taskSet.wait();
768 }
769 return;
770 }
771
772 auto chunkSizing =
773 detail::adjustChunkSizing(range, maxThreads, isStatic, minItemsPerChunk, N, options.wait);
774 maxThreads = chunkSizing.maxThreads;
775 isStatic = chunkSizing.isStatic;
776
777 // If adjustment reduced threads below 2, run inline — not worth parallelizing.
778 if (maxThreads < 2) {
779 detail::initStates(states, defaultState, 1, options.reuseExistingState);
780 f(*states.begin(), range.start, range.end);
781 if (options.wait) {
782 taskSet.wait();
783 }
784 return;
785 }
786
787 if (isStatic) {
788 detail::parallel_for_staticImpl(
789 taskSet,
790 states,
791 defaultState,
792 range,
793 std::forward<F>(f),
794 static_cast<ssize_t>(maxThreads),
795 options.wait,
796 options.reuseExistingState);
797 return;
798 }
799
800 const size_type numToLaunch = std::min<size_type>(maxThreads - options.wait, N);
801
802 detail::initStates(
803 states,
804 defaultState,
805 static_cast<size_t>(numToLaunch + options.wait),
806 options.reuseExistingState);
807
808 if (numToLaunch == 1 && !options.wait) {
809 taskSet.schedule(
810 [&s = states.front(), range, f = std::move(f)]() { f(s, range.start, range.end); });
811 return;
812 }
813
814 auto chunkInfo = range.calcChunkSize(numToLaunch, options.wait, minItemsPerChunk);
815 auto chunkSize = std::get<0>(chunkInfo);
816 auto numChunks = std::get<1>(chunkInfo);
817
818 if (options.wait) {
819 alignas(kCacheLineSize) std::atomic<decltype(numChunks)> index(0);
820 detail::parallel_for_dynamicImpl(
821 taskSet,
822 states,
823 range.start,
824 range.end,
825 std::forward<F>(f),
826 static_cast<size_t>(numToLaunch),
827 chunkSize,
828 numChunks,
829 index,
830 [](auto) {},
831 options.wait);
832 } else {
833 using SizeType = decltype(numChunks);
834 struct ChunkIndex {
835 std::atomic<SizeType> index;
836 };
837 static_assert(sizeof(ChunkIndex) <= kCacheLineSize, "ChunkIndex must fit in one cache line");
838 char* mem = allocSmallBuffer<kCacheLineSize>();
839 auto* ci = new (mem) ChunkIndex{{0}};
840 SizeType lastExit = numChunks + static_cast<SizeType>(numToLaunch) - 1;
841 detail::parallel_for_dynamicImpl(
842 taskSet,
843 states,
844 range.start,
845 range.end,
846 std::forward<F>(f),
847 static_cast<size_t>(numToLaunch),
848 chunkSize,
849 numChunks,
850 ci->index,
851 [ci, lastExit](auto cur) {
852 if (cur == lastExit) {
853 deallocSmallBuffer<kCacheLineSize>(ci);
854 }
855 },
856 options.wait);
857 }
858}
