76template <typename T>
77class ResourcePool {
78 public:
86 template <typename F>
87 ResourcePool(size_t size, const F& init)
88 : pool_(size),
89 backingResources_(reinterpret_cast<char*>(
90 detail::alignedMallocOrThrow(size * detail::alignToCacheLine(sizeof(T))))),
91 size_(size) {
92 char* buf = backingResources_;
93
94 // There are three reasons we create our own buffer and use placement new:
95 // 1. We want to be able to handle non-movable non-copyable objects
96 // * Note that we could do this with std::deque
97 // 2. We want to minimize memory allocations, since that can be a common point of contention in
98 // multithreaded programs.
99 // 3. We can easily ensure that the objects are cache aligned to help avoid false sharing.
100
101 // If init() or a T constructor throws, the destructor will not run, so release what has been
102 // built so far here.
103 struct Cleanup {
104 char* backing;
105 size_t constructed;
106 ~Cleanup() {
107 if (!backing) {
108 return;
109 }
110 for (size_t i = 0; i < constructed; ++i) {
111 reinterpret_cast<T*>(backing + i * detail::alignToCacheLine(sizeof(T)))->~T();
112 }
113 detail::alignedFree(backing);
114 }
115 } cleanup{backingResources_, 0};
116
117 while (cleanup.constructed < size) {
118 T* t = new (buf) T(init());
119 ++cleanup.constructed;
120 pool_.enqueue(t);
121 buf += detail::alignToCacheLine(sizeof(T));
122 }
123 cleanup.backing = nullptr;
124 }

131 Resource<T> acquire() {
132 T* t;
133 DISPENSO_TSAN_ANNOTATE_IGNORE_WRITES_BEGIN();
134 pool_.wait_dequeue(t);
135 DISPENSO_TSAN_ANNOTATE_IGNORE_WRITES_END();
136 return Resource<T>(t, this);
137 }

143 ~ResourcePool() {
144 assert(pool_.size_approx() == size_);
145 for (size_t i = 0; i < size_; ++i) {

148 pool_.wait_dequeue(t);

151 }
152 detail::alignedFree(backingResources_);
153 }
