283 template <typename F>
284 DISPENSO_REQUIRES(OnceCallableFunc<F>)
285 void schedule(
286 F&& f,
287 bool skipRecheck = false,
288 float poolRecursiveLoadFactor = kDefaultPoolRecursiveLoadFactor) {
289 if (cost_ == TaskCost::kHeavy) {
290 schedulePlaced(std::forward<F>(f), skipRecheck, poolRecursiveLoadFactor);
291 return;
292 }
293 // Combined inline decision (mirrors scheduleBulkImpl):
294 // 1. TaskSet-level overload
295 // 2. Pool-recursive: pool threads inline when workRemaining_ exceeds
296 // numThreads * poolRecursiveLoadFactor
297 // 3. Pool global: non-recursive callers inline at the loose poolLoadFactor_
298 // After this check, use ForceQueuingTag to skip the redundant check
299 // in ThreadPool::schedule.
300 if (outstandingTaskCount_.load(std::memory_order_relaxed) > taskSetLoadFactor_ &&
301 DISPENSO_EXPECT(!canceled(), true)) {
302 f();
303 return;
304 }
305 if (!skipRecheck) {
306 ssize_t curWork = pool_.workRemaining_.load(std::memory_order_relaxed);
307 ssize_t quickFactor =
308 static_cast<ssize_t>(static_cast<float>(pool_.numThreads()) * poolRecursiveLoadFactor);
309 if ((detail::PerPoolPerThreadInfo::isPoolRecursive(&pool_) && curWork > quickFactor) ||
310 curWork > pool_.poolLoadFactor_.load(std::memory_order_relaxed)) {
311 f();
312 return;
313 }
314 }
315 pool_.schedule(packageTask(std::forward<F>(f)), ForceQueuingTag());
316 }
