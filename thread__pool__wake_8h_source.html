29// Platform-tunable wake parameters.
30// BranchFactor: max children per node in the wake cascade tree.
31// Higher values = fewer cascade levels but more serial futex calls per level.
32// Override via -DDISPENSO_TUNE_WAKE_BRANCH_FACTOR=N for cross-platform tuning.
33#if defined(DISPENSO_TUNE_WAKE_BRANCH_FACTOR)
34constexpr int32_t kDefaultWakeBranchFactor = DISPENSO_TUNE_WAKE_BRANCH_FACTOR;
35#elif defined(_WIN32)
36// Windows: WaitOnAddress is more expensive per-call, so use wider fan-out
37// to reduce cascade depth.
38constexpr int32_t kDefaultWakeBranchFactor = 8;
39#else
40// Linux/macOS: futex is cheap per-call, so 4-wide gives the best
41// balance of cascade depth vs per-level cost.
42constexpr int32_t kDefaultWakeBranchFactor = 4;
43#endif

63static constexpr int32_t kMaxThreadsPerWakeGroup = 64;
64
65// Number of threads sharing one EpochWaiter (and thus one futex address).
66// A single futex_wake(INT_MAX) wakes all threads in a subgroup, reducing
67// the number of syscalls for batch wakes by this factor. The tradeoff is
68// that targeted wakes (cascade wakeOne) cause up to kWaiterSubgroupSize-1
69// spurious wakes — those threads check for work, find none, and sleep again.
70// Override via -DDISPENSO_TUNE_WAITER_SUBGROUP_SIZE=N for cross-platform tuning.
71#if defined(DISPENSO_TUNE_WAITER_SUBGROUP_SIZE)
72static constexpr int32_t kWaiterSubgroupSize = DISPENSO_TUNE_WAITER_SUBGROUP_SIZE;
73#else
74static constexpr int32_t kWaiterSubgroupSize = 4;
75#endif
76
77// distributeBudget declares EpochWaiter peerSlots[kWaiterSubgroupSize - 1];
78// a value of 1 would yield a zero-length array (UB).
79static_assert(kWaiterSubgroupSize >= 2, "kWaiterSubgroupSize must be at least 2");
80
81struct DISPENSO_CACHELINE_ALIGNED WaiterBlock {
82 detail::EpochWaiter waiters[kMaxThreadsPerWakeGroup / kWaiterSubgroupSize];
83};

146 void enterSleep(int32_t threadIdx) {
147 int32_t group = threadIdx / groupSize_;
148 int32_t bit = threadIdx % groupSize_;
149 groupStates_[static_cast<size_t>(group)].sleepMask.fetch_or(

160 void exitSleep(int32_t threadIdx) {
161 int32_t group = threadIdx / groupSize_;
162 int32_t bit = threadIdx % groupSize_;
163 groupStates_[static_cast<size_t>(group)].sleepMask.fetch_and(
164 ~(uint64_t{1} << bit), std::memory_order_relaxed);
165 totalSleeping_.fetch_sub(1, std::memory_order_relaxed);
166 }

175 bool tryClaimSleeper(int32_t threadIdx) {
176 int32_t group = threadIdx / groupSize_;
177 int32_t bit = threadIdx % groupSize_;
178 uint64_t bitMask = uint64_t{1} << bit;
179 uint64_t prev = groupStates_[static_cast<size_t>(group)].sleepMask.fetch_and(
180 ~bitMask, std::memory_order_acq_rel);
181 return (prev & bitMask) != 0;

273 int32_t processBudget(int32_t threadIdx) {
274 // Precomputed lookup: non-cascade threads have kNoCascade, cascade-team
275 // threads have their budget slot index. Avoids division/modulo per call.
276 uint16_t slot = budgetSlotTable_[static_cast<size_t>(threadIdx)];
277 if (slot == kNoCascade) {
278 return 0;
279 }
280 if (wakeBudgets_[slot].value.load(std::memory_order_relaxed) <= 0) {
281 return 0;
282 }
283 return processBudgetSlow(threadIdx);
