31// Platform-tunable wake parameters.
32// BranchFactor: max children per node in the wake cascade tree.
33// Higher values = fewer cascade levels but more serial futex calls per level.
34// Override via -DDISPENSO_TUNE_WAKE_BRANCH_FACTOR=N for cross-platform tuning.
35#if defined(DISPENSO_TUNE_WAKE_BRANCH_FACTOR)
36constexpr int32_t kDefaultWakeBranchFactor = DISPENSO_TUNE_WAKE_BRANCH_FACTOR;
37#elif defined(_WIN32)
38// Windows: WaitOnAddress is more expensive per-call, so use wider fan-out
39// to reduce cascade depth.
40constexpr int32_t kDefaultWakeBranchFactor = 8;
41#else
42// Linux/macOS: futex is cheap per-call, so 4-wide gives the best
43// balance of cascade depth vs per-level cost.
44constexpr int32_t kDefaultWakeBranchFactor = 4;
45#endif

228 // ---- Pattern C cascade for scheduleBulkToRings ----
229 //
230 // 1-seed 2-level static cascade tree:
231 // Producer wakes seed group (g0) with one bumpAndWakeN.
232 // g0's threads each have a cascade-host slot; on wake they execute a
233 // lambda that wakes one level-2 target group, then runs their own user
234 // work.
235 // level-2 groups' threads each have cascade-host slots that wake one
236 // level-3 target group.
237 //
238 // Tree shape for N threads (groupSize=G):
239 // level 1: 1 group (g0) covers G threads
240 // level 2: G groups covers G*G threads
241 // level 3: G*G groups (when needed) covers G*G*G threads
242 //
243 // 192 threads / G=8 = 24 groups: levels 1+2 cover 1+8 = 9 groups (72
244 // threads). Need level 3 for the remaining 15 groups.
245
251 int32_t cascadeTargetFor(int32_t threadIdx, int32_t count) const {
252 if (threadIdx < 0 || static_cast<size_t>(threadIdx) >= cascadeTargets_.size()) {
253 return -1;
254 }
255 int32_t target = cascadeTargets_[static_cast<size_t>(threadIdx)];
256 int32_t lastGroup = (count - 1) / groupSize_;
257 return (target <= lastGroup) ? target : -1;
258 }
259
262 void cascadeWake(int32_t targetGroup) {
263 uint64_t mask =
264 groupStates_[static_cast<size_t>(targetGroup)].sleepMask.load(std::memory_order_relaxed);
265 auto& waiter = waiterFor(targetGroup * groupSize_);
266 if (mask == 0) {
267 waiter.bump();
268 } else {
269 int32_t numSleepers = detail::countSetBits(mask);
270 waiter.bumpAndWakeN(numSleepers, groupSize_);
271 }
272 }
