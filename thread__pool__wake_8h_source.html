29// Platform-tunable wake parameters.
30// BranchFactor: max children per node in the wake cascade tree.
31// Higher values = fewer cascade levels but more serial futex calls per level.
32// Override via -DDISPENSO_TUNE_WAKE_BRANCH_FACTOR=N for cross-platform tuning.
33#if defined(DISPENSO_TUNE_WAKE_BRANCH_FACTOR)
34constexpr int32_t kDefaultWakeBranchFactor = DISPENSO_TUNE_WAKE_BRANCH_FACTOR;
35#elif defined(_WIN32)
36// Windows: WaitOnAddress is more expensive per-call, so use wider fan-out
37// to reduce cascade depth.
38constexpr int32_t kDefaultWakeBranchFactor = 8;
39#else
40// Linux/macOS: futex is cheap per-call, so 4-wide gives the best
41// balance of cascade depth vs per-level cost.
42constexpr int32_t kDefaultWakeBranchFactor = 4;
43#endif

57struct DISPENSO_CACHELINE_ALIGNED GroupWakeState {
58 std::atomic<uint64_t> sleepMask{0};
59 std::atomic<uint64_t> wakePending{0};
60 // Promote seed group: index of a second seed group set by wakeN when the
61 // cascade tree from one seed alone can't claim all wakePending bits in
62 // parallel (numCascadeGroups > groupSize). The first awoken thread of
63 // this group atomically exchanges to -1 and bumpAndWakes the promote
64 // target, bootstrapping parallel cascade from two seeds with one
65 // producer syscall. Sentinel -1 = no promote pending.
66 std::atomic<int32_t> promoteSeed{-1};
67};

77// One EpochWaiter (futex address) per group: all threads in a group share
78// it. A single futex_wake(addr, K) wakes K arbitrary threads from the group;
79// futex_wake(addr, 1) wakes one. Bulk wakes for K > groupSize fan out
80// across groups via the wakePending bitmask cascade (see GroupWakeState
81// below) — no waitv needed, workers wait on a single address via plain
82// futex_wait.
83struct DISPENSO_CACHELINE_ALIGNED WaiterBlock {
84 detail::EpochWaiter waiter;
85};

147 void enterSleep(int32_t threadIdx) {

149 int32_t bit = threadIdx % groupSize_;

161 void exitSleep(int32_t threadIdx) {
162 int32_t group = threadIdx / groupSize_;
163 int32_t bit = threadIdx % groupSize_;
164 groupStates_[static_cast<size_t>(group)].sleepMask.fetch_and(
165 ~(uint64_t{1} << bit), std::memory_order_relaxed);
166 totalSleeping_.fetch_sub(1, std::memory_order_relaxed);
167 }

176 bool tryClaimSleeper(int32_t threadIdx) {
177 int32_t group = threadIdx / groupSize_;
178 int32_t bit = threadIdx % groupSize_;
179 uint64_t bitMask = uint64_t{1} << bit;
180 uint64_t prev = groupStates_[static_cast<size_t>(group)].sleepMask.fetch_and(
181 ~bitMask, std::memory_order_acq_rel);
182 return (prev & bitMask) != 0;
183 }
