31// Platform-tunable wake parameters.
32// BranchFactor: max children per node in the wake cascade tree.
33// Higher values = fewer cascade levels but more serial futex calls per level.
34// Override via -DDISPENSO_TUNE_WAKE_BRANCH_FACTOR=N for cross-platform tuning.
35#if defined(DISPENSO_TUNE_WAKE_BRANCH_FACTOR)
36constexpr int32_t kDefaultWakeBranchFactor = DISPENSO_TUNE_WAKE_BRANCH_FACTOR;
37#elif defined(_WIN32)
38// Windows: WaitOnAddress is more expensive per-call, so use wider fan-out
39// to reduce cascade depth.
40constexpr int32_t kDefaultWakeBranchFactor = 8;
41#else
42// Linux/macOS: futex is cheap per-call, so 4-wide gives the best
43// balance of cascade depth vs per-level cost.
44constexpr int32_t kDefaultWakeBranchFactor = 4;
45#endif

67// One EpochWaiter (futex address) per group: all threads in a group share
68// it. A single futex_wake(addr, K) wakes K arbitrary threads from the
69// group; futex_wake(addr, 1) wakes one. Bulk wakes for K > groupSize fan
70// out across groups via the Pattern C cascade — workers in the seed group
71// each carry a pre-staged lambda that wakes one target group; level-2
72// cascade hosts in those groups wake the remaining groups in parallel.
73struct DISPENSO_CACHELINE_ALIGNED WaiterBlock {
74 detail::EpochWaiter waiter;
75};

225 // ---- Pattern C cascade for scheduleBulkToRings ----
226 //
227 // 1-seed 2-level static cascade tree:
228 // Producer wakes seed group (g0) with one bumpAndWakeN.
229 // g0's threads each have a cascade-host slot; on wake they execute a
230 // lambda that wakes one level-2 target group, then runs their own user
231 // work.
232 // level-2 groups' threads each have cascade-host slots that wake one
233 // level-3 target group.
234 //
235 // Tree shape for N threads (groupSize=G):
236 // level 1: 1 group (g0) covers G threads
237 // level 2: G groups covers G*G threads
238 // level 3: G*G groups (when needed) covers G*G*G threads
239 //
240 // 192 threads / G=8 = 24 groups: levels 1+2 cover 1+8 = 9 groups (72
241 // threads). Need level 3 for the remaining 15 groups.
242
243 // Returns the level-2/3 target group for a cascade-host thread, or -1 if
244 // this thread is not a cascade host OR if the target group lies outside
245 // the wake range [0, count). The count gate matches scheduleBulkToRings's
246 // "wake only threads with work" contract: for partial-pool dispatch we
247 // don't want cascade hosts firing wakes into empty groups.
248 int32_t cascadeTargetFor(int32_t threadIdx, int32_t count) const {
249 if (threadIdx < 0 || static_cast<size_t>(threadIdx) >= cascadeTargets_.size()) {
250 return -1;
251 }
252 int32_t target = cascadeTargets_[static_cast<size_t>(threadIdx)];
253 int32_t lastGroup = (count - 1) / groupSize_;
254 return (target <= lastGroup) ? target : -1;
255 }
256
257 // Wakes one target group: bumps the group's epoch and issues bumpAndWakeN
258 // only if the sleepMask is non-zero. Called by cascade-host lambdas.
259 void cascadeWake(int32_t targetGroup) {
260 uint64_t mask =
261 groupStates_[static_cast<size_t>(targetGroup)].sleepMask.load(std::memory_order_relaxed);
262 auto& waiter = waiterFor(targetGroup * groupSize_);
263 if (mask == 0) {
264 waiter.bump();
265 } else {
266 int32_t numSleepers = detail::countSetBits(mask);
267 waiter.bumpAndWakeN(numSleepers, groupSize_);
268 }
269 }
